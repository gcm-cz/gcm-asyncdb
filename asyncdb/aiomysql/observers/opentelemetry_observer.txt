from typing import Any, Collection

from opentelemetry.trace import Span, SpanKind, StatusCode, Tracer

from asyncdb.aiomysql import Transaction, TransactionObserver
from asyncdb.exceptions import Error


class OpenTelemetryTransactionObserver(TransactionObserver):
    def __init__(self, tracer: Tracer) -> None:
        """
        :param tracer: OpenTelemetry tracer to create spans with.
        """
        self.tracer = tracer

    def observe_transaction_start(self, transaction: Transaction, **kwargs: Any) -> Span:
        """Start a span for the transaction."""
        with self.tracer.start_as_current_span(
            name="Transaction",
            kind=SpanKind.CLIENT,
            end_on_exit=False,
            attributes={
                "server.address": transaction.connection.host,
                "server.port": transaction.connection.port,
                "db.namespace": transaction.connection.db,
            },
        ) as span:
            return span

    def observe_transaction_commit(self, transaction: Transaction, context: Span, **kwargs: Any) -> None:
        """End the transaction span."""
        context.set_status(StatusCode.OK, "COMMIT")

    def observe_transaction_rollback(self, transaction: Transaction, context: Span, **kwargs: Any) -> None:
        """End the transaction span with error status."""
        context.set_status(StatusCode.OK, "ROLLBACK")

    def observe_transaction_end(self, transaction: Transaction, context: Span, **kwargs: Any) -> None:
        """End the transaction span."""
        context.end()

    def observe_query_before(self, transaction: Transaction, query: str, trx_context: Span, **kwargs: Any) -> Span:
        """Add query information to the transaction span."""
        with self.tracer.start_as_current_span(
            name="SQL Query",
            kind=SpanKind.CLIENT,
            attributes={
                "db.query.text": query,
            },
            end_on_exit=False,
        ) as span:
            return span

    def observe_query_after(
        self,
        transaction: Transaction,
        query: str,
        context: Span,
        trx_context: Span,
        **kwargs: Any,
    ) -> None:
        """End the query span."""
        context.set_status(StatusCode.OK)
        context.end()

    def observe_query_error(
        self,
        transaction: Transaction,
        query: str,
        error: BaseException,
        context: Span,
        trx_context: Span,
        *,
        query_pattern: str,
        query_attrs: Collection[Any],
        **kwargs: Any,
    ) -> None:
        """End the query span with error status."""
        if isinstance(error, Error):
            context.set_attribute("db.response.status_code", error.code)

        context.set_attributes(
            {
                "error.type": type(error).__name__,
            }
        )

        context.record_exception(error)
        context.set_status(StatusCode.ERROR, str(error))
        context.end()
